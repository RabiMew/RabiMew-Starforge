// Starforge first-install keybind defaults (Default Options mod).
// Format: key_<mapping name>:<InputConstants key name>[:MODIFIER]
// Applied only to bindings still at factory default / never user-set.
// Player-remapped keys are never overwritten on pack updates.
//
// Layout contract (full table + conflict audit: docs/keybinds.md):
//   In-game cluster ZXCVGHB: TaCZ refit Z, crawl C, zoom/melee V,
//   fire-select G, inspect H, backpack B; reload R, gun-interact O,
//   quick-switch-gun X. GUI-only functions may reuse these letters —
//   they cannot fire while playing (inventory sort R, recipes R/U...).
//   Alt+key combos host low-frequency toggles.

// === Guide / progression ===
// FTB Quests book -> J (main guide); ProgressiveStages tree -> K.
key_key.ftbquests.quests:key.keyboard.j
key_key.progressivestages.open_tree:key.keyboard.k
// KubeJS Kubedex is a demo/dev GUI -> unbound (was K).
key_key.kubejs.kubedex:key.keyboard.unknown

// === Xaero map cluster ===
// M world map, Y minimap settings, N new waypoint, U waypoint list.
// Upstream B (new waypoint) stays off B (backpack). Waypoint list back on
// upstream U: in-game only, JEI/EMI Uses-U is GUI-only -> context-exclusive.
key_gui.xaero_open_map:key.keyboard.m
key_gui.xaero_minimap_settings:key.keyboard.y
key_gui.xaero_waypoints_key:key.keyboard.u
key_gui.xaero_new_waypoint:key.keyboard.n
// Enlarged-minimap hold key (was Z) unbound: Z belongs to TaCZ refit,
// and M already opens the full-screen map.
key_gui.xaero_enlarge_map:key.keyboard.unknown
// World-map settings key (]) duplicates minimap settings on Y; unbound.
key_gui.xaero_open_settings:key.keyboard.unknown

// === TaCZ combat cluster (upstream layout kept) ===
// reload R / refit Z / crawl C / zoom V + melee V (zoom applies only while
// aiming, melee only while not aiming -> upstream-shared V is safe) /
// fire-select G / inspect H / interact O / shoot LMB / aim RMB / config Alt+T.
// TaCZ-addon "Quickly Switch Gun" ships on Tab -> collides with the vanilla
// player list (real in-game conflict); moved to X.
key_key.taczaddon.switch_gun.desc:key.keyboard.x
// Sophisticated Backpacks "run interaction upgrades" was C -> collides with
// TaCZ crawl in-game; moved to Alt+C (low-frequency combo).
key_key.sophisticatedbackpacks.inventory_interaction:key.keyboard.c:ALT

// === Vanilla creative hotbar save/load -> unbound ===
// Frees C (TaCZ crawl) and X (quick-switch); niche creative-only feature.

// === Iris shader keys off combat letters ===
